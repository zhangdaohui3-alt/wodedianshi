440 |         if (context == null || snapshot == null) throw new IllegalArgumentException("catalogue snapshot is null");
441 |         HashSet<String> favoriteKeys = null; // Preserve favorites atomically inside writeRecords().
442 |         // Channel count is deliberately not compared with the previous catalogue:
443 |         // a fully collected and normalized refresh may correctly remove dead,
444 |         // duplicate or invalid channels. Completeness is validated structurally

448 |             throw new IllegalStateException("Refusing an empty background catalogue snapshot");
449 |         }

456 |             }

459 |             channel.url = item.optString("url", "").trim();
460 |             channel.group = item.optString("group", "");
